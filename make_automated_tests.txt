# =============================================================================
# clone_manual_test_folder_to_auto.ps1
#
# 1. Walks the source Xray Test Repository folder (SOURCE_FOLDER_NAME),
#    preserving its full subfolder structure
# 2. Creates a mirrored destination folder (DEST_FOLDER_NAME) at the same level
# 3. For each Manual test in each subfolder, clones it as a new Generic/
#    Automated test in the matching destination subfolder
#    (originals are NEVER modified)
# 4. Updates AccelQ scenario requirement traceability to include the new keys
#
# Fill in the variables below before running.
# =============================================================================

# ── User-configurable variables ───────────────────────────────────────────────

$JIRA_DOMAIN = "a.jira.org"          # Jira hostname only (no https://)
$JIRA_PAT = "aaaaaaaaaaaaa"       # Jira Personal Access Token
$TARGET_PROJECT = "SRCWEBRED"           # Jira project key

# Folder names exactly as they appear in the Xray Test Repository UI
$SOURCE_FOLDER_NAME = "CBC"
$DEST_FOLDER_NAME = "CBC (Automated)"

# Subfolders to skip (at any depth). Use just a name to skip all folders with
# that name anywhere in the tree, or a path like "Parent/Child" to target only
# that specific folder. Children of skipped folders are also skipped.
$IGNORE_FOLDERS = @(
    "Index Page",
    "Lengthy Tags",
    "Supports Date Filtering on CBC Release",
    "Table Design for CBC",
    "Lengthy Tags/Articles List Page"
)

$ACCELQ_EMAIL = "a@a.com"
$ACCELQ_DOMAIN = "accelq.example.pvt"
$ACCELQ_API_KEY = "aaaaaaaaaaaaaa"
$ACCELQ_PROJECT = "a_project"
$ACCELQ_TENANT = "aaa"

# ── Auth headers ──────────────────────────────────────────────────────────────

$jiraHeaders = @{
    "Authorization" = "Bearer $JIRA_PAT"
    "Accept"        = "application/json"
    "Content-Type"  = "application/json"
}

$accelqHeaders = @{
    "user_id"      = $ACCELQ_EMAIL
    "api_key"      = $ACCELQ_API_KEY
    "Accept"       = "application/json"
    "Content-Type" = "application/json"
}

# ── Log file ──────────────────────────────────────────────────────────────────
$LogFile = Join-Path $PSScriptRoot ("clone_log_" + (Get-Date -Format "yyyy-MM-dd_HH-mm-ss") + ".txt")
Start-Transcript -Path $LogFile
Write-Host "Logging to: $LogFile"

# ── Step 1: Discover custom field IDs ────────────────────────────────────────

Write-Host "Discovering Xray custom field IDs..."
$allFields = Invoke-RestMethod -Uri "https://$JIRA_DOMAIN/rest/api/2/field" -Headers $jiraHeaders
$testTypeField = $allFields | Where-Object { $_.name -eq "Test Type" } | Select-Object -First 1
$runTypeField = $allFields | Where-Object { $_.name -eq "Run Type" } | Select-Object -First 1

if (-not $testTypeField) { Write-Error "Cannot find 'Test Type' custom field."; exit 1 }
if (-not $runTypeField) { Write-Error "Cannot find 'Run Type' custom field."; exit 1 }

$testTypeFieldId = $testTypeField.id
$runTypeFieldId = $runTypeField.id
Write-Host "  Test Type : $testTypeFieldId"
Write-Host "  Run Type  : $runTypeFieldId"

# ── Step 2: Find Test issue type ID ──────────────────────────────────────────

$projectMeta = Invoke-RestMethod -Uri "https://$JIRA_DOMAIN/rest/api/2/issue/createmeta?projectKeys=$TARGET_PROJECT&expand=projects.issuetypes" -Headers $jiraHeaders
$testIssueType = $projectMeta.projects[0].issuetypes | Where-Object { $_.name -eq "Test" } | Select-Object -First 1

if (-not $testIssueType) { Write-Error "Cannot find 'Test' issue type in project $TARGET_PROJECT."; exit 1 }
$testIssueTypeId = $testIssueType.id
Write-Host "  Test issue type ID: $testIssueTypeId"

# ── Helper: find a folder by name and return it with its parent's ID ──────────

function Find-FolderAndParent {
    param($treeNode, [string]$targetName)
    foreach ($child in $treeNode.folders) {
        if ($child.name -eq $targetName) {
            return @{ folder = $child; parentId = $treeNode.id }
        }
        if ($child.folders -and $child.folders.Count -gt 0) {
            $result = Find-FolderAndParent -treeNode $child -targetName $targetName
            if ($result) { return $result }
        }
    }
    return $null
}

# ── Helper: get all tests in a folder (paged) ────────────────────────────────

function Get-FolderTests {
    param([string]$folderId)
    $allTests = @()
    $page = 1
    $batchSize = 50
    do {
        $result = Invoke-RestMethod `
            -Uri     "https://$JIRA_DOMAIN/rest/raven/1.0/api/testrepository/$TARGET_PROJECT/folders/$folderId/tests?page=$page&limit=$batchSize" `
            -Headers $jiraHeaders
        if ($result.tests) { $allTests += $result.tests }
        $page++
    } while ($allTests.Count -lt $result.total)
    return $allTests
}

# ── Helper: clone tests in a folder, then recurse into subfolders ─────────────

function Clone-FolderContents {
    param($srcFolder, [string]$destFolderId, [string]$indent = "  ", [string]$currentPath = "")

    $folderTests = Get-FolderTests -folderId ([string]$srcFolder.id)
    $newKeysHere = @()

    foreach ($test in $folderTests) {
        $oldKey = $test.key

        try {
            $issue = Invoke-RestMethod `
                -Uri     "https://$JIRA_DOMAIN/rest/api/2/issue/$oldKey" `
                -Headers $jiraHeaders
        }
        catch {
            Write-Host "${indent}[ERR] Could not fetch $oldKey : $($_.Exception.Message)"
            $script:failCount++
            continue
        }

        $newIssueBody = @{
            fields = @{
                project                 = @{ key = $TARGET_PROJECT }
                issuetype               = @{ id = $script:testIssueTypeId }
                summary                 = $issue.fields.summary
                $script:testTypeFieldId = @{ value = "Generic" }
                $script:runTypeFieldId  = @{ value = "Automated" }
            }
        }
        if ($issue.fields.description) {
            $newIssueBody.fields["description"] = $issue.fields.description
        }
        if ($issue.fields.labels -and $issue.fields.labels.Count -gt 0) {
            $newIssueBody.fields["labels"] = $issue.fields.labels
        }

        try {
            $created = Invoke-RestMethod `
                -Uri     "https://$JIRA_DOMAIN/rest/api/2/issue" `
                -Method  Post `
                -Headers $jiraHeaders `
                -Body    ($newIssueBody | ConvertTo-Json -Depth 10)

            $newKey = $created.key
            $script:keyMap[$oldKey] = $newKey
            $newKeysHere += $newKey
            Write-Host "${indent}[OK]  $oldKey -> $newKey  ($($issue.fields.summary))"

            # Copy requirement links
            $reqLinks = $issue.fields.issuelinks | Where-Object {
                $_.type.name -eq "Tests" -and $_.outwardIssue
            }
            foreach ($link in $reqLinks) {
                $reqKey = $link.outwardIssue.key
                $linkBody = @{
                    type         = @{ name = "Tests" }
                    outwardIssue = @{ key = $reqKey }
                    inwardIssue  = @{ key = $newKey }
                } | ConvertTo-Json
                try {
                    Invoke-RestMethod -Uri "https://$JIRA_DOMAIN/rest/api/2/issueLink" -Method Post -Headers $jiraHeaders -Body $linkBody | Out-Null
                    Write-Host "${indent}       Linked: $reqKey"
                }
                catch {
                    Write-Host "${indent}       [WARN] Could not link $reqKey : $($_.Exception.Message)"
                }
            }

            $script:successCount++
        }
        catch {
            $status = $_.Exception.Response.StatusCode.value__
            Write-Host "${indent}[ERR] $oldKey -> $status : $($_.Exception.Message)"
            $script:failCount++
        }
    }

    # Place all cloned tests into the destination folder
    if ($newKeysHere.Count -gt 0) {
        try {
            Invoke-RestMethod `
                -Uri     "https://$JIRA_DOMAIN/rest/raven/1.0/api/testrepository/$TARGET_PROJECT/folders/$destFolderId/tests" `
                -Method  Put `
                -Headers $jiraHeaders `
                -Body    (@{ add = $newKeysHere } | ConvertTo-Json) | Out-Null
        }
        catch {
            Write-Host "${indent}[WARN] Could not add tests to folder ${destFolderId}: $($_.Exception.Message)"
        }
    }

    # Recurse into subfolders, mirroring the structure
    if ($srcFolder.folders -and $srcFolder.folders.Count -gt 0) {
        foreach ($child in $srcFolder.folders) {
            $childPath = if ($currentPath) { "$currentPath/$($child.name)" } else { $child.name }
            # Skip if any ignore entry matches the folder name alone OR a suffix of the current path
            $shouldSkip = $script:IGNORE_FOLDERS | Where-Object {
                $childPath -eq $_ -or $childPath -like "*/$_"
            }
            if ($shouldSkip) {
                Write-Host "${indent}[SKIP] $childPath"
                continue
            }
            Write-Host "${indent}Folder: $($child.name)"
            try {
                $newChild = Invoke-RestMethod `
                    -Uri     "https://$JIRA_DOMAIN/rest/raven/1.0/api/testrepository/$TARGET_PROJECT/folders/$destFolderId" `
                    -Method  Post `
                    -Headers $jiraHeaders `
                    -Body    (@{ name = $child.name } | ConvertTo-Json)
                Clone-FolderContents -srcFolder $child -destFolderId ([string]$newChild.id) -indent ("  " + $indent) -currentPath $childPath
            }
            catch {
                Write-Host "${indent}[ERR] Could not create subfolder '$($child.name)': $($_.Exception.Message)"
            }
        }
    }
}

# ── Step 3: Get folder hierarchy and locate source folder ─────────────────────

Write-Host "`nFetching Test Repository folder hierarchy..."
$hierarchy = Invoke-RestMethod `
    -Uri     "https://$JIRA_DOMAIN/rest/raven/1.0/api/testrepository/$TARGET_PROJECT/folders" `
    -Headers $jiraHeaders

$found = Find-FolderAndParent -treeNode $hierarchy -targetName $SOURCE_FOLDER_NAME
if (-not $found) {
    Write-Error "Cannot find folder '$SOURCE_FOLDER_NAME' in the Test Repository for project $TARGET_PROJECT."
    exit 1
}
$srcFolder = $found.folder
$srcParentId = $found.parentId
Write-Host "  Found '$SOURCE_FOLDER_NAME' (id: $($srcFolder.id), parent id: $srcParentId)"

# ── Step 4: Create destination folder at the same level as source ─────────────

Write-Host "`nCreating destination folder '$DEST_FOLDER_NAME'..."
try {
    $destFolder = Invoke-RestMethod `
        -Uri     "https://$JIRA_DOMAIN/rest/raven/1.0/api/testrepository/$TARGET_PROJECT/folders/$srcParentId" `
        -Method  Post `
        -Headers $jiraHeaders `
        -Body    (@{ name = $DEST_FOLDER_NAME } | ConvertTo-Json)
    Write-Host "  Created '$DEST_FOLDER_NAME' (id: $($destFolder.id))"
}
catch {
    $status = $_.Exception.Response.StatusCode.value__
    Write-Error "Could not create destination folder ($status): $($_.Exception.Message)"
    exit 1
}

# ── Step 5: Walk source folder tree, clone tests into mirrored structure ───────

Write-Host "`n── Cloning '$SOURCE_FOLDER_NAME' -> '$DEST_FOLDER_NAME' ─────────────────"
$successCount = 0
$failCount = 0
$keyMap = @{}

Clone-FolderContents -srcFolder $srcFolder -destFolderId ([string]$destFolder.id)

Write-Host "`nDone. $successCount cloned, $failCount failed."
Write-Host "Original Manual tests are untouched."

if ($keyMap.Count -eq 0) {
    Write-Host "`nNo successful clones - skipping AccelQ traceability update."
    Stop-Transcript
    exit 0
}

# ── Step 6: Update AccelQ scenario traceability ───────────────────────────────

Write-Host "`n── Updating AccelQ requirement traceability ──────────────────────────────"

$ACCELQ_SCN_BASE = "https://$ACCELQ_DOMAIN/awb/api/2.0/$ACCELQ_TENANT/$ACCELQ_PROJECT/scn/scenarios/auto"
$oldKeys = $keyMap.Keys | ForEach-Object { $_ }

$accelqScenarios = @()
$accelqPageNum = 1
$accelqPageSize = 25

do {
    $fetchBody = @{
        filter       = @{ requirementKeys = $oldKeys }
        fetchDetails = @{ requirementTraceability = $true }
    } | ConvertTo-Json -Depth 5

    $fetchResult = Invoke-RestMethod `
        -Uri     "$ACCELQ_SCN_BASE/fetch-details?pageNumber=$accelqPageNum&pageSize=$accelqPageSize" `
        -Method  Post `
        -Headers $accelqHeaders `
        -Body    $fetchBody

    $accelqScenarios += $fetchResult.results
    $accelqPageNum++
} while ($accelqScenarios.Count -lt $fetchResult.totalCount)

Write-Host "Found $($accelqScenarios.Count) AccelQ scenario(s) to update."

# Auto-discover the externalIntegrationProject value from existing traceability
$integrationProject = $accelqScenarios `
| ForEach-Object { $_.requirementTraceability } `
| Where-Object { $_ -and $keyMap.ContainsKey($_.key) } `
| Select-Object -ExpandProperty externalIntegrationProject -First 1

if (-not $integrationProject) {
    Write-Host "  [ERR] Could not auto-detect AccelQ integration project name from existing scenario traceability."
    Write-Host "        Ensure AccelQ scenarios are already linked to the old Manual Jira keys."
    exit 1
}
Write-Host "  Detected integration project: $integrationProject"

$scenarioNewKeys = @{}

foreach ($scn in $accelqScenarios) {
    $scnId = $scn.id
    if (-not $scenarioNewKeys.ContainsKey($scnId)) {
        $scenarioNewKeys[$scnId] = [System.Collections.Generic.List[string]]::new()
    }
    $linkedOldKeys = $scn.requirementTraceability `
    | Where-Object { $keyMap.ContainsKey($_.key) } `
    | Select-Object -ExpandProperty key
    foreach ($oldKey in $linkedOldKeys) {
        $newKey = $keyMap[$oldKey]
        if (-not $scenarioNewKeys[$scnId].Contains($newKey)) {
            $scenarioNewKeys[$scnId].Add($newKey)
        }
    }
}

$grouped = @{}
foreach ($scnId in $scenarioNewKeys.Keys) {
    $keySetStr = ($scenarioNewKeys[$scnId] | Sort-Object) -join ","
    if (-not $grouped.ContainsKey($keySetStr)) {
        $grouped[$keySetStr] = [System.Collections.Generic.List[int]]::new()
    }
    $grouped[$keySetStr].Add($scnId)
}

$bulkPayload = @()
foreach ($keySetStr in $grouped.Keys) {
    $bulkPayload += @{
        scenarioIds             = @($grouped[$keySetStr])
        requirementTraceability = @{
            requirementKeys            = @($keySetStr -split ",")
            updateType                 = "APPEND"
            externalIntegrationProject = $integrationProject
        }
    }
}

try {
    $bulkResult = Invoke-RestMethod `
        -Uri     "$ACCELQ_SCN_BASE/bulk-update" `
        -Method  Patch `
        -Headers $accelqHeaders `
        -Body    ($bulkPayload | ConvertTo-Json -Depth 10)

    Write-Host "  Updated scenario IDs  : $($bulkResult.updatedEntities -join ', ')"
    if ($bulkResult.notUpdatedOthersCheckedOutEntities.Count -gt 0) {
        Write-Host "  [WARN] Checked-out (not updated): $($bulkResult.notUpdatedOthersCheckedOutEntities -join ', ')"
        Write-Host "         These scenarios are checked out by another user in AccelQ. Update them manually."
    }
    if ($bulkResult.notFoundEntities.Count -gt 0) {
        Write-Host "  [WARN] Not found: $($bulkResult.notFoundEntities -join ', ')"
    }
}
catch {
    $status = $_.Exception.Response.StatusCode.value__
    Write-Host "  [ERR] AccelQ bulk-update failed ($status): $($_.Exception.Message)"
}

Write-Host "`nAll done."
Stop-Transcript
