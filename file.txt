window._batchLogs = [];

const origFetch = window.fetch;
window.fetch = async function(...args) {
    const req = args[0] instanceof Request ? args[0] : new Request(...args);
    if (req.url.includes('$batch')) {
        const clone = req.clone();
        clone.text().then(body => {
            window._batchLogs.push(body);
            console.log('Captured batch #' + window._batchLogs.length);
        });
    }
    return origFetch.apply(this, args);
};







copy(window._batchLogs.join('\n\n=== NEXT BATCH ===\n\n'));
