param (
    [Parameter(Mandatory=$true, Position=0)]
    [string]$InputPath,

    [Parameter(Mandatory=$false, Position=1)]
    [string]$OutputPath
)

try {
    # Resolve absolute path for Outlook COM
    $ResolvedInput = (Resolve-Path -Path $InputPath -ErrorAction Stop).Path

    # Default to saving as .txt with the same name in the same directory
    if (-not $OutputPath) {
        $OutputPath = [System.IO.Path]::ChangeExtension($ResolvedInput, ".txt")
    } else {
        $OutputPath = [System.IO.Path]::GetFullPath($OutputPath)
    }

    # Initialize Outlook COM
    $outlook = New-Object -ComObject Outlook.Application
    $msg = $outlook.CreateItemFromTemplate($ResolvedInput)

    # Format email header & body into plain text
    $content = @(
        "Subject: $($msg.Subject)",
        "From:    $($msg.SenderName) <$($msg.SenderEmailAddress)>",
        "To:      $($msg.To)",
        "CC:      $($msg.CC)",
        "Date:    $($msg.SentOn)",
        "--------------------------------------------------------------------------------",
        "",
        $msg.Body
    ) -join "`r`n"

    # Save to file using UTF-8 encoding
    [System.IO.File]::WriteAllText($OutputPath, $content, [System.Text.Encoding]::UTF8)

    # Clean up Outlook COM objects
    $msg = $null
    $outlook.Quit()
    [System.Runtime.Interopservices.Marshal]::ReleaseComObject($outlook) | Out-Null
    [System.GC]::Collect()

    Write-Host "Successfully saved text file to: $OutputPath" -ForegroundColor Green
}
catch {
    Write-Error "Failed to convert .msg file: $_"
    exit 1
}
